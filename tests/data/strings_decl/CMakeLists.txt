# strings_declaration_* (spec 029 T5.7): cmake/nnh_strings.cmake's checks of LANGUAGES, BASE and the string files, one
# case per configure (-DCASE=...). Only file names matter: nothing is read or built.
cmake_minimum_required(VERSION 3.24)
project(nnh_strings_declaration NONE)
include(${NNH_ROOT}/cmake/nnh_strings.cmake)
add_custom_target(game)
set(languages en de)
set(base en)
set(files strings/en.strings strings/de.strings images/hero.png)
if(CASE STREQUAL "missing")
    set(files strings/en.strings)
elseif(CASE STREQUAL "unlisted")
    list(APPEND files strings/fr.strings)
elseif(CASE STREQUAL "bad_code")
    set(languages en PT_br)
elseif(CASE STREQUAL "both_forms")
    list(APPEND files strings/de/menus.strings)
elseif(CASE STREQUAL "bad_base")
    set(base ja)
elseif(CASE STREQUAL "no_languages")
    set(languages "")
endif()
set_target_properties(game PROPERTIES NNH_LANGUAGES "${languages}" NNH_BASE_LANGUAGE "${base}")
_nnh_strings_files(game "${files}" "${CMAKE_CURRENT_SOURCE_DIR}/assets" strings)
get_target_property(sources game NNH_STRING_SOURCES)
message("strings declaration: ${sources}")
